> > ## 🎨 PART 2: Dynamic Identity System
> >
> > ### ThemeContext Implementation
> >
> > \`\`\`typescript
> > // contexts/ThemeContext.tsx
> >
> > import { createContext, useContext, useState, useEffect, ReactNode } from 'react';
> >
> > type AppCategory = 'dating' | 'finance' | 'tech' | 'productivity' | 'creative';
> > type AccentColor = 'rose' | 'emerald' | 'cyan' | 'amber' | 'purple';
> >
> > interface ThemeContextType {
> >   appCategory: AppCategory;
> >   accentColor: AccentColor;
> >   setAppCategory: (category: AppCategory) => void;
> > }
> >
> > const ThemeContext = createContext<ThemeContextType | undefined>(undefined);
> >
> > const categoryAccentMap: Record<AppCategory, AccentColor> = {
> >   dating: 'rose',
> >   finance: 'emerald',
> >   tech: 'cyan',
> >   productivity: 'amber',
> >   creative: 'purple'
> > };
> >
> > export function ThemeProvider({ children, initialCategory = 'tech' }: {
> >   children: ReactNode;
> >   initialCategory?: AppCategory;
> > }) {
> >   const [appCategory, setAppCategory] = useState<AppCategory>(initialCategory);
> >   const [accentColor, setAccentColor] = useState<AccentColor>(
> >     categoryAccentMap[initialCategory]
> >   );
> >
> >   useEffect(() => {
> >     // Update accent color when category changes
> >     const newAccent = categoryAccentMap[appCategory];
> >     setAccentColor(newAccent);
> >
> >     // Update CSS custom property
> >     document.documentElement.style.setProperty(
> >       '--accent-primary',
> >       `var(--accent-${newAccent})`
> >     );
> >   }, [appCategory]);
> >
> >   return (
> >     <ThemeContext.Provider value={{ appCategory, accentColor, setAppCategory }}>
> >       {children}
> >     </ThemeContext.Provider>
> >   );
> > }
> >
> > export function useTheme() {
> >   const context = useContext(ThemeContext);
> >   if (!context) {
> >     throw new Error('useTheme must be used within ThemeProvider');
> >   }
> >   return context;
> > }
> > \`\`\`
> >
> > ### Usage Example
> >
> > \`\`\`typescript
> > // app/layout.tsx
> >
> > import { ThemeProvider } from '@/contexts/ThemeContext';
> >
> > export default function RootLayout({ children }: { children: React.ReactNode }) {
> >   return (
> >     <html lang="en">

        <ThemeProvider initialCategory="tech">
                  {children}

      </body>
          </html>
            );
> > }

> >

  );
